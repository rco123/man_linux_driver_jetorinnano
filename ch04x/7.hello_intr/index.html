GPIO 인터럽터

헤드핀의 40핀에 대하여 인터럽터 동작을 확인하여 본다. PI.00에 해당하는 GPIO 번호를 확인하고, LOW → HIGH 변화를 감지하여 인터럽트를 발생시키는 커널 모듈작성한다. 이 코드는 request_irq()를 사용하여 인터럽트 핸들러를 등록하고, gpio_to_irq(399)을 통해 GPIO 399번에 대한 인터럽트를 설정합니다.커널 모듈을 작성하여 삽입 시 ON, 제거 시 OFF하도록 하는 과정을 실행합니다.

1. GPIO 번호 확인

1) 헤드핀의 핀 이름 확인

Jetson Orin Nano Developer Kit Carrier Board Specification ( SP-11324-001_v1.3 )에서 핀에 대한 정보를 확인한다.

40번 핀에 연결된 물리적 핀의 이름은 PI.00 으로 되어 있다. 40번 핀에 ON OFF를 할수 있는 와이어점퍼를 연결한다..

2) 할당된 GPIO 번호 확인

sudo cat /sys/kernel/debug/gpio
이 명령을 실행하면 현재 시스템에서 사용 가능한 GPIO 리스트가 출력됩니다. PI.00에 할당된 GPIO 번호를 확인해야 합니다.

예를 들어, PI.00이 gpiochip0의 399번 핀이라면, 해당 GPIO 번호는 399이 됩니다.

2. 커널 모듈 작성

아래 커널 모듈은 gpio_number 변수에 확인한 GPIO 번호를 설정하여, 모듈을 삽입하면 인터럽터가 되도록 설정합니다.

🔹 hello.c 작성

#include <linux/module.h>
#include <linux/gpio.h>
#include <linux/interrupt.h>
#include <linux/irq.h>

#define GPIO_NUM 399

static int irq_number;

static irqreturn_t gpio_irq_handler(int irq, void *dev_id)
{
    int gpio_state = gpio_get_value(GPIO_NUM); // GPIO 상태 확인
    pr_info("GPIO %d Interrupt Occurred! (LOW -> HIGH), state: %d\n", GPIO_NUM, gpio_state);
    return IRQ_HANDLED;
}

static int __init hello_init(void)
{
    int result;

    if (!gpio_is_valid(GPIO_NUM)) {
        pr_err("Invalid GPIO %d\n", GPIO_NUM);
        return -ENODEV;
    }

    result = gpio_request(GPIO_NUM, "gpio_399");
    if (result) {
        pr_err("Failed to request GPIO %d\n", GPIO_NUM);
        return result;
    }

    result = gpio_direction_input(GPIO_NUM);
    if (result) {
        pr_err("Failed to set GPIO %d as input\n", GPIO_NUM);
        gpio_free(GPIO_NUM);
        return result;
    }

    irq_number = gpio_to_irq(GPIO_NUM);
    if (irq_number < 0) {
        pr_err("Failed to get IRQ for GPIO %d\n", GPIO_NUM);
        gpio_free(GPIO_NUM);
        return irq_number;
    }

    pr_info("GPIO %d assigned to IRQ %d\n", GPIO_NUM, irq_number);

    result = request_irq(irq_number, gpio_irq_handler, IRQF_TRIGGER_FALLING | IRQF_TRIGGER_RISING, "gpio_399_irq", NULL);
    if (result) {
        pr_err("Failed to request IRQ %d for GPIO %d\n", irq_number, GPIO_NUM);
        gpio_free(GPIO_NUM);
        return result;
    }

    pr_info("GPIO %d interrupt initialized successfully\n", GPIO_NUM);
    return 0;
}

static void __exit hello_exit(void)
{
    free_irq(irq_number, NULL);
    gpio_free(GPIO_NUM);
    pr_info("GPIO %d interrupt module removed\n", GPIO_NUM);
}

module_init(hello_init);
module_exit(hello_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("Your Name");
MODULE_DESCRIPTION("GPIO 399 Interrupt Handler (LOW to HIGH)");

3. 모듈 컴파일 및 실행

🔹 Makefile 작성

obj-m += hello.o

KDIR := /lib/modules/$(shell uname -r)/build
PWD  := $(shell pwd)

all:
    $(MAKE) -C $(KDIR) M=$(PWD) modules

clean:
    $(MAKE) -C $(KDIR) M=$(PWD) clean

🔹 컴파일

make

🔹 모듈 로드

sudo insmod hello.ko

🔹 모듈 제거

sudo rmmod hello

🔹 로그 확인

dmesg | tail -n 20
