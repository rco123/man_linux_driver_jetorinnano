드라이브 기동

dtb 정보의 compatible 의 매칭을 통한 드라이브 자동 기동을 수행한다.

1. 등록 코드

static const struct of_device_id my_device_dt_match[] = {
    { .compatible = "my_led_run" },
    { /* sentinel */ }
};

2. 프로그램 작성

#include <linux/module.h>
#include <linux/platform_device.h>
#include <linux/of.h>

static int my_device_probe(struct platform_device *pdev) {
    struct device *dev = &pdev->dev;
    struct device_node *np = dev->of_node;
    const char *label;
    int reg, pin, active_state;

    pr_info("my_device_probe: Device found!\n");

    if (!np) {
        pr_err("No device tree node found!\n");
        return -EINVAL;
    }

    // Device Tree 속성 값 읽기
    if (of_property_read_string(np, "label", &label) == 0)
        pr_info("Label: %s\n", label);

    if (of_property_read_u32(np, "reg", &reg) == 0)
        pr_info("Register: %d\n", reg);

    if (of_property_read_u32(np, "pin", &pin) == 0)
        pr_info("Pin: %d\n", pin);

    if (of_property_read_u32(np, "active_state", &active_state) == 0)
        pr_info("Active State: %d\n", active_state);

    return 0;
}

static int my_device_remove(struct platform_device *pdev) {
    pr_info("my_device_remove: Device unregistered\n");
    return 0;
}

static const struct of_device_id my_device_dt_match[] = {
    { .compatible = "my_led_run" },
    { /* sentinel */ }
};

MODULE_DEVICE_TABLE(of, my_device_dt_match);

static struct platform_driver my_device_driver = {
    .driver = {
        .name = "my_device",
        .of_match_table = my_device_dt_match,
    },
    .probe = my_device_probe,
    .remove = my_device_remove,
};

module_platform_driver(my_device_driver);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("Your Name");
MODULE_DESCRIPTION("Simple Platform Driver");
