시스템콜

리눅스에서 시스템콜을 직접 호출하여 커널 기능을 사용하는 코드를 작성할 수 있습니다. 일반적으로 시스템콜은 C 언어를 통해 호출하며, syscall() 함수를 사용하거나 인라인 어셈블리를 통해 직접 호출할 수 있습니다.

1. 구현항목

1) 시스템 콜 함수 구현

asmlinkage long __arm64_sys_my_systemcall(const struct pt_regs *regs)
{
    int a, b, result;
    printk(KERN_INFO "my_systemcall invoked\n");

    // Your custom logic here
    a = regs->regs[0]; // First argument (a)
    b = regs->regs[1]; // Second argument (b)
    result = a * b;    // Multiply a and b

    printk(KERN_INFO "my_systemcall invoked: a = %d, b = %d, result = %d\n", a, b, result);

    return result; // Return the result

}
  • ARM64 아키텍처에서는 시스템 콜의 첫 6개 인자가 x0부터 x5 레지스터를 통해 전달됩니다. 이는 pt_regs 구조체의 regs[0]부터 regs[5]에 해당합니다.

2) 시스템 콜 데이블 업데이트

/*
 * Wrappers to pass the pt_regs argument.
 */
#define __arm64_sys_personality     __arm64_sys_arm64_personality

#undef __SYSCALL
#define __SYSCALL(nr, sym)  asmlinkage long __arm64_##sym(const struct pt_regs *);
#include <asm/unistd.h>

#undef __SYSCALL
#define __SYSCALL(nr, sym)  [nr] = __arm64_##sym,


const syscall_fn_t sys_call_table[__NR_syscalls] = {
    [0 ... __NR_syscalls - 1] = __arm64_sys_ni_syscall,

#include <asm/unistd.h>
    [440] = __arm64_sys_my_systemcall,
};

2) 케널 재 컴파일 후 업로드

make ARCH=arm64 LOCALVERSION=-tegra CROSS_COMPILE=${CROSS_COMPILE_AARCH64} -j8 --output-sync=target V=1 Image
jetson 보드의 /boot/Image 를 생성된 Image 로 교체한다.

3) 테스트 코드 작성

syscall.c 작성

#include <unistd.h>
#include <sys/syscall.h>
#include <stdio.h>

#define __NR_my_systemcall 440 // unistd.h에서 정의한 시스템 콜 번호

int main() {
    int a = 5;
    int b = 10;
    long result = syscall(__NR_my_systemcall, a, b);

    printf("시스템 콜 반환값: %ld\n", result); // 50 (5 * 10) 출력
    return 0;
}

4) 컴파일 및 실행

gcc syscall.c -o syscall
./syscall
