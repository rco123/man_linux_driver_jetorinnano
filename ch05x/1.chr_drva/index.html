문자 드라이브 I

Jetson Orin Nano에서 캐릭터 디바이스 드라이버(char driver)를 작성하는 과정과 샘플 코드를 설명하겠습니다.
이 과정에서는 디바이스 드라이버 작성 → 커널 모듈 빌드 → 상위 애플리케이션 작성 → 테스트 단계를 진행합니다.

1. 캐릭터 디바이스 드라이버 개요

캐릭터 디바이스 드라이버는 /dev/mydevice와 같은 인터페이스를 제공하여 read(), write() 같은 표준 시스템 콜을 통해 데이터를 주고받을 수 있도록 합니다.

주요 작업

  1. 메이저 번호, 마이너 번호 할당
  2. 메이저 번호: 디바이스 유형을 구분
  3. 마이너 번호: 동일한 메이저 번호 내에서 여러 장치를 구분
  4. 파일 연산 등록 (read(), write())
  5. open(), release(), read(), write() 등의 함수 정의
  6. 디바이스 파일 생성 (mknod 사용)
  7. 커널 모듈 빌드 및 삽입 (insmod, rmmod)
  8. 상위 프로그램에서 open(), read(), write() 실행

2. Jetson Orin Nano용 캐릭터 디바이스 드라이버 코드

아래 코드는 Jetson Orin Nano에서 사용할 간단한 캐릭터 디바이스 드라이버입니다.

1) char_driver.c - 캐릭터 드라이버 코드**

#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/uaccess.h>

#define DEVICE_NAME "mychar"
#define BUF_SIZE 1024

static dev_t dev_num;
static struct cdev my_cdev;
static char device_buffer[BUF_SIZE];

static int my_open(struct inode *inode, struct file *file) {
    printk(KERN_INFO "mychar: Device opened\n");
    return 0;
}

static int my_release(struct inode *inode, struct file *file) {
    printk(KERN_INFO "mychar: Device closed\n");
    return 0;
}

static ssize_t my_read(struct file *file, char __user *user_buffer, size_t size, loff_t *offset) {
    int bytes_read = size < BUF_SIZE ? size : BUF_SIZE;
    if (copy_to_user(user_buffer, device_buffer, bytes_read)) {
        return -EFAULT;
    }
    printk(KERN_INFO "mychar: Read %d bytes\n", bytes_read);
    return bytes_read;
}

static ssize_t my_write(struct file *file, const char __user *user_buffer, size_t size, loff_t *offset) {
    int bytes_write = size < BUF_SIZE ? size : BUF_SIZE;
    if (copy_from_user(device_buffer, user_buffer, bytes_write)) {
        return -EFAULT;
    }
    printk(KERN_INFO "mychar: Written %d bytes\n", bytes_write);
    return bytes_write;
}

static struct file_operations fops = {
    .owner = THIS_MODULE,
    .open = my_open,
    .release = my_release,
    .read = my_read,
    .write = my_write,
};

static int __init mychar_init(void) {
    if (alloc_chrdev_region(&dev_num, 0, 1, DEVICE_NAME) < 0) {
        printk(KERN_ALERT "mychar: Failed to allocate device number\n");
        return -1;
    }
    cdev_init(&my_cdev, &fops);
    if (cdev_add(&my_cdev, dev_num, 1) < 0) {
        unregister_chrdev_region(dev_num, 1);
        printk(KERN_ALERT "mychar: Failed to add cdev\n");
        return -1;
    }
    printk(KERN_INFO "mychar: Device initialized with major %d, minor %d\n", MAJOR(dev_num), MINOR(dev_num));
    return 0;
}

static void __exit mychar_exit(void) {
    cdev_del(&my_cdev);
    unregister_chrdev_region(dev_num, 1);
    printk(KERN_INFO "mychar: Device unregistered\n");
}

module_init(mychar_init);
module_exit(mychar_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("Jetson Orin Nano Dev");
MODULE_DESCRIPTION("Simple Character Device Driver for Jetson Orin Nano");

2) Makefile - 빌드 스크립트**

obj-m := char_driver.o

KDIR := /lib/modules/$(shell uname -r)/build
PWD := $(shell pwd)

all:
    make -C $(KDIR) M=$(PWD) modules

clean:
    make -C $(KDIR) M=$(PWD) clean
  • obj-m := char_driver.o → 모듈을 빌드할 파일 지정
  • KDIR := /lib/modules/$(shell uname -r)/build → 현재 커널 헤더 디렉토리 지정

3. 커널 모듈 빌드 및 로드

1) 빌드 

make
2) 커널 모듈 로드 
sudo insmod char_driver.ko

3) 디바이스 파일 생성

sudo dmesg | tail -n 10
sudo mknod /dev/mychar c <major_number> 0
sudo chmod 666 /dev/mychar
- <major_number>는 sudo dmesg | grep "mychar" 실행 후 확인 가능

4) 커널 모듈 언로드 

sudo rmmod char_driver

4. 상위 프로그램 작성

1) test_app.c - 상위 사용자 애플리케이션 

#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>

#define DEVICE "/dev/mychar"

int main() {
    int fd;
    char write_buf[] = "Hello Jetson Orin Nano!";
    char read_buf[1024];

    fd = open(DEVICE, O_RDWR);
    if (fd < 0) {
        perror("Failed to open device");
        return -1;
    }

    write(fd, write_buf, strlen(write_buf));
    printf("Written to device: %s\n", write_buf);

    read(fd, read_buf, sizeof(read_buf));
    printf("Read from device: %s\n", read_buf);

    close(fd);
    return 0;
}

2) 컴파일 및 실행 

gcc test_app.c -o test_app
./test_app

3) 디바이스 파일 생성 
sudo mknod /dev/mychar c <major_number> 0
sudo chmod 666 /dev/mychar
4) 상위 프로그램 실행 
./test_app

6. 결론

이 실습을 통해 Jetson Orin Nano에서 캐릭터 디바이스 드라이버를 작성하고, read()와 write()를 테스트하는 방법을 배웠습니다.
이제 실제 하드웨어 장치와 연동하여 센서 데이터를 읽거나 특정 기능을 수행하는 드라이버로 확장할 수 있습니다. 🚀
