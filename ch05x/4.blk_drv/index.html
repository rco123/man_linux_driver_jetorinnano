블록 드라이버 (Block Driver) 개요

블록 드라이버(blk 드라이버)는 블록 단위로 데이터를 읽고 쓰는 디바이스 드라이버입니다. 일반적으로 디스크 드라이버와 같은 저장 장치에서 사용됩니다. 리눅스에서 블록 디바이스는 read() 및 write() 요청을 블록 단위로 처리하며, 캐싱 및 버퍼링을 통해 성능을 향상시킵니다.

이제 간단한 블록 디바이스 드라이버를 작성하고, 이를 등록하여 상위 애플리케이션에서 사용할 수 있도록 해보겠습니다.

1. 블록 디바이스 드라이버 코드 작성

이 드라이버는 다음 기능을 포함합니다.

  • 고정된 메이저 번호 사용 (240 사용)
  • 노드 생성 (mknod 사용)
  • 입력된 데이터를 내부 메모리에 저장 후 반환
  • 기본적인 open, release, read, write 동작 구현

블록 디바이스 드라이버 코드

아래는 간단한 블록 디바이스 드라이버 코드입니다.

#include <linux/module.h>
#include <linux/fs.h>
#include <linux/init.h>
#include <linux/kernel.h>
#include <linux/slab.h>
#include <linux/uaccess.h>

#define DEVICE_NAME "blkdev"
#define BLOCK_MAJOR 240
#define BLOCK_SIZE  512

static char *device_buffer;

static int blk_open(struct inode *inode, struct file *filp) {
    printk(KERN_INFO "blkdev: Device opened\n");
    return 0;
}

static int blk_release(struct inode *inode, struct file *filp) {
    printk(KERN_INFO "blkdev: Device closed\n");
    return 0;
}

static ssize_t blk_read(struct file *filp, char __user *buffer, size_t length, loff_t *offset) {
    size_t to_read = min((size_t)(BLOCK_SIZE - *offset), length);

    if (to_read == 0) {
        printk(KERN_INFO "blkdev: Read end of buffer\n");
        return 0;
    }

    if (copy_to_user(buffer, device_buffer + *offset, to_read)) {
        return -EFAULT;
    }

    *offset += to_read;
    printk(KERN_INFO "blkdev: Read %zu bytes\n", to_read);
    return to_read;
}

static ssize_t blk_write(struct file *filp, const char __user *buffer, size_t length, loff_t *offset) {
    size_t to_write = min((size_t)(BLOCK_SIZE - *offset), length);

    if (to_write == 0) {
        printk(KERN_INFO "blkdev: Write end of buffer\n");
        return -ENOSPC;
    }

    if (copy_from_user(device_buffer + *offset, buffer, to_write)) {
        return -EFAULT;
    }

    *offset += to_write;
    printk(KERN_INFO "blkdev: Written %zu bytes\n", to_write);
    return to_write;
}

static struct file_operations blk_fops = {
    .owner   = THIS_MODULE,
    .open    = blk_open,
    .release = blk_release,
    .read    = blk_read,
    .write   = blk_write
};

static int __init blk_init(void) {
    int ret;

    ret = register_chrdev(BLOCK_MAJOR, DEVICE_NAME, &blk_fops);
    if (ret < 0) {
        printk(KERN_ERR "blkdev: Failed to register device\n");
        return ret;
    }

    device_buffer = kmalloc(BLOCK_SIZE, GFP_KERNEL);
    if (!device_buffer) {
        unregister_chrdev(BLOCK_MAJOR, DEVICE_NAME);
        return -ENOMEM;
    }

    memset(device_buffer, 0, BLOCK_SIZE);
    printk(KERN_INFO "blkdev: Module loaded, Major number: %d\n", BLOCK_MAJOR);
    return 0;
}

static void __exit blk_exit(void) {
    kfree(device_buffer);
    unregister_chrdev(BLOCK_MAJOR, DEVICE_NAME);
    printk(KERN_INFO "blkdev: Module unloaded\n");
}

module_init(blk_init);
module_exit(blk_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("ChatGPT");
MODULE_DESCRIPTION("Simple Block Device");

2. 블록 디바이스 드라이버 등록 및 노드 생성

1) 드라이버 빌드 및 로드

위 코드를 blkdev.c로 저장한 후 Makefile을 생성하여 컴파일합니다.

Makefile

obj-m += blkdev.o

all:
    make -C /lib/modules/$(shell uname -r)/build M=$(PWD) modules

clean:
    make -C /lib/modules/$(shell uname -r)/build M=$(PWD) clean

빌드 및 설치

make
sudo insmod blkdev.ko

2) 블록 디바이스 노드 생성

sudo mknod /dev/blkdev c 240 0
sudo chmod 666 /dev/blkdev

3. 블록 디바이스를 사용하는 사용자 애플리케이션 작성

이제 상위 애플리케이션을 작성하여 블록 디바이스를 읽고 쓸 수 있도록 합니다.

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>

#define DEVICE "/dev/blkdev"

int main() {
    int fd;
    char write_buf[] = "Hello, Block Device!";
    char read_buf[512];

    fd = open(DEVICE, O_RDWR);
    if (fd < 0) {
        perror("Failed to open device");
        return -1;
    }

    printf("Writing to device: %s\n", write_buf);
    write(fd, write_buf, strlen(write_buf));

    lseek(fd, 0, SEEK_SET);  // 버퍼의 시작으로 이동

    read(fd, read_buf, sizeof(read_buf));
    printf("Read from device: %s\n", read_buf);

    close(fd);
    return 0;
}

빌드 및 실행

gcc -o blk_test blk_test.c
./blk_test

4. 블록 디바이스 드라이버 제거

테스트가 끝난 후, 모듈을 제거하려면 다음 명령을 실행합니다.

sudo rmmod blkdev
sudo rm /dev/blkdev

요약

  1. 블록 디바이스 드라이버 (blkdev.c) 작성
  2. 모듈 빌드 (Makefile 사용)
  3. 블록 디바이스 등록 (insmod, mknod 사용)
  4. 사용자 애플리케이션 (blk_test.c) 작성하여 테스트
  5. 드라이버 언로드 (rmmod 및 /dev/blkdev 제거)

이제 이 블록 디바이스 드라이버를 리눅스 환경에서 직접 빌드하고 실행하여 테스트해볼 수 있습니다. 🚀
