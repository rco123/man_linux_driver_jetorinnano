mmap 메모리사용

커널 모듈을 이용한 공유 메모리 생성 및 접근에 관한 내용으로 사용자 공간에서 shm_open()을 사용하지 않고, 커널 모듈을 통해 공유 메모리를 생성하고 여러 프로세스가 이를 접근하도록 만들 수 있습니다. 이 과정은 다음 단계로 진행됩니다.

  • 커널 모듈을 작성하여 공유 메모리를 생성 (kmalloc 또는 vmalloc 이용 가능)
  • mmap()을 통해 사용자 공간과 커널 공간 간 메모리 매핑 제공
  • 두 개의 사용자 애플리케이션이 이를 활용하여 데이터 공유 (하나는 쓰기, 하나는 읽기)

1. 커널 모듈 (shared_mem.ko) 작성

커널 모듈에서는 kmalloc()을 이용해 공유 메모리를 할당하고, mmap()을 통해 사용자 공간에서 접근할 수 있도록 합니다.

📌 커널 모듈 코드 (shared_mem.c)

#include <linux/init.h>
#include <linux/module.h>
#include <linux/fs.h>
#include <linux/slab.h>
#include <linux/mm.h>
#include <linux/uaccess.h>

#define DEVICE_NAME "shared_mem"
#define SHARED_MEM_SIZE 4096
#define MAJOR_NUMBER 500 // 고정된 Major Number

static char *shared_memory;

static int device_open(struct inode *inode, struct file *file) {
    return 0;
}

static int device_release(struct inode *inode, struct file *file) {
    return 0;
}

static int device_mmap(struct file *filp, struct vm_area_struct *vma) {
    unsigned long size = vma->vm_end - vma->vm_start;

    if (size > SHARED_MEM_SIZE) {
        return -EINVAL;
    }

    if (remap_pfn_range(vma, vma->vm_start,
                        virt_to_phys(shared_memory) >> PAGE_SHIFT,
                        size, vma->vm_page_prot)) {
        return -EAGAIN;
    }

    return 0;
}

static struct file_operations fops = {
    .open = device_open,
    .release = device_release,
    .mmap = device_mmap,
};

static int __init shared_mem_init(void) {
    shared_memory = kmalloc(SHARED_MEM_SIZE, GFP_KERNEL);
    if (!shared_memory) {
        pr_err("Failed to allocate shared memory\n");
        return -ENOMEM;
    }

    if (register_chrdev(MAJOR_NUMBER, DEVICE_NAME, &fops) < 0) {
        pr_err("Failed to register device\n");
        kfree(shared_memory);
        return -EBUSY;
    }

    pr_info("Shared memory module loaded. Major: %d\n", MAJOR_NUMBER);
    return 0;
}

static void __exit shared_mem_exit(void) {
    unregister_chrdev(MAJOR_NUMBER, DEVICE_NAME);
    kfree(shared_memory);
    pr_info("Shared memory module unloaded\n");
}

module_init(shared_mem_init);
module_exit(shared_mem_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("ChatGPT");
MODULE_DESCRIPTION("Kernel Module for Shared Memory");

2. 커널 모듈 빌드

1) 📌 Makefile을 작성하여 모듈을 컴파일합니다.

obj-m += shared_mem.o

all:
    make -C /lib/modules/$(shell uname -r)/build M=$(PWD) modules

clean:
    make -C /lib/modules/$(shell uname -r)/build M=$(PWD) clean

2) 빌드 및 모듈 로드 

make
sudo insmod shared_mem.ko

3. 사용자 공간 프로그램 작성

1) 📌 공유 메모리에 쓰기 (writer.c) 

#include <stdio.h>
#include <fcntl.h>
#include <sys/mman.h>
#include <unistd.h>
#include <string.h>

#define DEVICE_PATH "/dev/shared_mem"
#define SHARED_MEM_SIZE 4096

int main() {
    int fd = open(DEVICE_PATH, O_RDWR);
    if (fd < 0) {
        perror("open");
        return 1;
    }

    // 공유 메모리 매핑
    char *ptr = mmap(NULL, SHARED_MEM_SIZE, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (ptr == MAP_FAILED) {
        perror("mmap");
        return 1;
    }

    // 데이터 쓰기
    strcpy(ptr, "Hello from writer!");
    printf("Writer: Data written to shared memory.\n");

    // 종료
    munmap(ptr, SHARED_MEM_SIZE);
    close(fd);
    return 0;
}

2) 📌 공유 메모리에서 읽기 (reader.c) 

#include <stdio.h>
#include <fcntl.h>
#include <sys/mman.h>
#include <unistd.h>

#define DEVICE_PATH "/dev/shared_mem"
#define SHARED_MEM_SIZE 4096

int main() {
    int fd = open(DEVICE_PATH, O_RDONLY);
    if (fd < 0) {
        perror("open");
        return 1;
    }

    // 공유 메모리 매핑
    char *ptr = mmap(NULL, SHARED_MEM_SIZE, PROT_READ, MAP_SHARED, fd, 0);
    if (ptr == MAP_FAILED) {
        perror("mmap");
        return 1;
    }

    // 데이터 읽기
    printf("Reader: Read from shared memory: %s\n", ptr);

    // 종료
    munmap(ptr, SHARED_MEM_SIZE);
    close(fd);
    return 0;
}

5. 실행 방법

  1. 커널 모듈 빌드 및 삽입 

    make
sudo insmod shared_mem.ko

  2. 디바이스 파일 생성 

    sudo mknod /dev/shared_mem c 500 0
sudo chmod 666 /dev/shared_mem

    💡 major_number는 dmesg | tail -n 10을 통해 확인 가능.

  3. Writer 실행 (쓰기) 

    gcc writer.c -o writer
./writer
Writer: Data written to shared memory.

  4. Reader 실행 (읽기)

    gcc reader.c -o reader
./reader
Reader: Read from shared memory: Hello from writer!

  5. 모듈 제거 

    sudo rmmod shared_mem

6. 정리

  • 커널 모듈을 통해 공유 메모리를 생성하고, mmap()을 통해 사용자 공간에 매핑
  • writer.c가 데이터를 쓰고, reader.c가 이를 읽음
  • mknod를 통해 /dev/shared_mem 디바이스 파일을 생성하여 접근 가능

이제, 이를 활용하여 IPC(프로세스 간 통신) 또는 장치 드라이버 개발 등에 확장할 수 있습니다! 🚀
