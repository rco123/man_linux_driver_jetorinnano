문자 드라이브 III

1. struct file_operations 설명

struct file_operations는 **리눅스 커널의 캐릭터 디바이스 드라이버**에서 파일과 관련된 시스템 콜을 처리하는 함수 포인터 구조체입니다. 이 구조체를 통해 open(), read(), write() 같은 동작을 구현할 수 있습니다.

기본적인 구조

static struct file_operations fops = {
    .owner = THIS_MODULE,  // 현재 모듈 소유권 설정
    .open = my_open,       // 디바이스 파일이 열릴 때 실행
    .release = my_release, // 디바이스 파일이 닫힐 때 실행
    .read = my_read,       // 사용자 공간에서 데이터를 읽을 때 실행
    .write = my_write,     // 사용자 공간에서 데이터를 쓸 때 실행
};

2. 추가 가능한 파일 연산자

아래 함수들을 추가하여 **디바이스의 기능을 확장**할 수 있습니다.

함수명 설명
.llseek 파일의 읽기/쓰기 위치를 변경 (파일 포인터 이동)
.mmap 사용자 공간과 커널 공간 간 메모리 매핑
.unlocked_ioctl IOCTL 시스템 콜을 처리하여 장치에 명령 전달
.poll select(), poll(), epoll() 시스템 콜을 위한 이벤트 감지
.fasync 비동기 알림 설정
.compat_ioctl 32비트 애플리케이션의 IOCTL 요청 처리 (64비트 커널에서 사용)

3. 확장된 file_operations 구조

추가적인 기능을 포함한 예제입니다.

static long my_ioctl(struct file *file, unsigned int cmd, unsigned long arg) {
    printk(KERN_INFO "mychar: IOCTL Command Received: %u\n", cmd);
    return 0;
}

static struct file_operations fops = {
    .owner = THIS_MODULE,
    .open = my_open,
    .release = my_release,
    .read = my_read,
    .write = my_write,
    .llseek = default_llseek,
    .unlocked_ioctl = my_ioctl, // IOCTL 처리
    .poll = my_poll,            // select(), poll() 등 지원
};

4. 플랫폼 디바이스를 통한 자동 등록 (DTB 사용)

Jetson Orin Nano에서 **디바이스 트리(DTB)**를 이용하여 문자 디바이스를 자동 등록하는 방법을 설명합니다.

1) 디바이스 트리(DTB)란?** - 하드웨어 정보를 설명하는 파일 (.dts) - 커널 부팅 시 장치 정보를 읽고 자동으로 드라이버와 매칭

5. 디바이스 트리 노드 작성

Jetson Orin Nano의 **디바이스 트리(DTB)**에 문자 디바이스를 등록하려면 dts 파일을 수정해야 합니다.

1) mychar_device.dts 예제 아래와 같은 디바이스 트리 노드를 추가합니다. 

/ {
    mychar: mychar@0 {
        compatible = "nvidia,mychar"; // 드라이버와 매칭할 문자열
        reg = <0x0 0x1000>;           // 장치 메모리 영역
        status = "okay";
    };
};
설명

  • compatible = "nvidia,mychar": 드라이버와 매칭할 문자열
  • reg = <0x0 0x1000>: I/O 메모리 영역
  • status = "okay": 활성화 상태

6. 플랫폼 드라이버 코드

플랫폼 드라이버를 작성하여 DTB에 등록된 디바이스를 자동 매칭하도록 합니다.

1) char_platform_driver.c - 플랫폼 드라이버 코드 

#include <linux/module.h>
#include <linux/platform_device.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/uaccess.h>

#define DEVICE_NAME "mychar"
#define MY_MAJOR 240
#define BUF_SIZE 1024

static struct cdev my_cdev;
static dev_t dev_num;
static struct class *my_class;
static struct device *my_device;
static char device_buffer[BUF_SIZE];

/* Open function */
static int my_open(struct inode *inode, struct file *file) {
    printk(KERN_INFO "mychar: Device opened\n");
    return 0;
}

/* Release function */
static int my_release(struct inode *inode, struct file *file) {
    printk(KERN_INFO "mychar: Device closed\n");
    return 0;
}

/* Read function */
static ssize_t my_read(struct file *file, char __user *user_buffer, size_t size, loff_t *offset) {
    int bytes_read = size < BUF_SIZE ? size : BUF_SIZE;
    if (copy_to_user(user_buffer, device_buffer, bytes_read)) {
        return -EFAULT;
    }
    printk(KERN_INFO "mychar: Read %d bytes\n", bytes_read);
    return bytes_read;
}

/* Write function */
static ssize_t my_write(struct file *file, const char __user *user_buffer, size_t size, loff_t *offset) {
    int bytes_write = size < BUF_SIZE ? size : BUF_SIZE;
    if (copy_from_user(device_buffer, user_buffer, bytes_write)) {
        return -EFAULT;
    }
    printk(KERN_INFO "mychar: Written %d bytes\n", bytes_write);
    return bytes_write;
}

/* File operations */
static struct file_operations fops = {
    .owner = THIS_MODULE,
    .open = my_open,
    .release = my_release,
    .read = my_read,
    .write = my_write,
};

/* Probe function: 디바이스가 등록될 때 실행 */
static int mychar_probe(struct platform_device *pdev) {
    int ret;

    printk(KERN_INFO "mychar: Platform device detected!\n");

    dev_num = MKDEV(MY_MAJOR, 0);
    ret = register_chrdev_region(dev_num, 1, DEVICE_NAME);
    if (ret < 0) {
        printk(KERN_ALERT "Failed to register device\n");
        return ret;
    }

    cdev_init(&my_cdev, &fops);
    ret = cdev_add(&my_cdev, dev_num, 1);
    if (ret < 0) {
        unregister_chrdev_region(dev_num, 1);
        return ret;
    }

    my_class = class_create(THIS_MODULE, DEVICE_NAME);
    if (IS_ERR(my_class)) {
        cdev_del(&my_cdev);
        unregister_chrdev_region(dev_num, 1);
        return PTR_ERR(my_class);
    }

    my_device = device_create(my_class, NULL, dev_num, NULL, DEVICE_NAME);
    if (IS_ERR(my_device)) {
        class_destroy(my_class);
        cdev_del(&my_cdev);
        unregister_chrdev_region(dev_num, 1);
        return PTR_ERR(my_device);
    }

    return 0;
}

/* Remove function */
static int mychar_remove(struct platform_device *pdev) {
    device_destroy(my_class, dev_num);
    class_destroy(my_class);
    cdev_del(&my_cdev);
    unregister_chrdev_region(dev_num, 1);
    printk(KERN_INFO "mychar: Platform device removed\n");
    return 0;
}

/* Platform driver structure */
static const struct of_device_id mychar_dt_ids[] = {
    { .compatible = "nvidia,mychar", },
    { }
};

MODULE_DEVICE_TABLE(of, mychar_dt_ids);

static struct platform_driver mychar_driver = {
    .probe = mychar_probe,
    .remove = mychar_remove,
    .driver = {
        .name = DEVICE_NAME,
        .of_match_table = mychar_dt_ids,
    },
};

module_platform_driver(mychar_driver);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("Jetson Orin Nano Dev");
MODULE_DESCRIPTION("Character Device Driver with Platform Device");

7. 결과 및 동작 방식

  1. 커널 부팅 시 → DTB에서 "nvidia,mychar"을 검색
  2. 매칭된 경우 → mychar_probe() 실행
  3. 자동으로 /dev/mychar 생성
  4. 사용자가 open(), read(), write() 실행 가능

이제 Jetson Orin Nano에서 DTB 기반으로 문자 디바이스를 자동으로 등록하고 사용할 수 있습니다. 🚀
