네트워크 드라이브 II

이제 커널 모듈이 **30초마다 인터럽트를 발생**하도록 타이머를 추가했습니다.
snull_timer_function()이 주기적으로 실행되며, snull_interrupt()를 호출하여 상위 네트워크 계층으로 패킷을 전송합니다.

1. 모듈작성

인터럽터를 자체 발생하여 패킷을 상부로 올리는 드라이브를 작성

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/netdevice.h>
#include <linux/etherdevice.h>
#include <linux/skbuff.h>
#include <linux/ip.h>
#include <linux/tcp.h>
#include <linux/udp.h>
#include <linux/interrupt.h>
#include <linux/timer.h>

static struct net_device *snull_dev;
static struct timer_list snull_timer;

static irqreturn_t snull_interrupt(int irq, void *dev_id) {
    struct net_device *dev = (struct net_device *)dev_id;
    struct sk_buff *skb;
    struct iphdr *iph;

    skb = netdev_alloc_skb(dev, 1500);
    if (!skb)
        return IRQ_HANDLED;

    skb_put(skb, 1500);
    skb->protocol = eth_type_trans(skb, dev);
    iph = ip_hdr(skb);

    printk(KERN_INFO "[snull] Packet received via interrupt\n");
    netif_rx(skb);

    return IRQ_HANDLED;
}

static void snull_timer_function(struct timer_list *t) {
    printk(KERN_INFO "[snull] Triggering interrupt manually\n");
    snull_interrupt(0, snull_dev);
    mod_timer(&snull_timer, jiffies + msecs_to_jiffies(30000));
}

static int snull_open(struct net_device *dev) {
    printk(KERN_INFO "[snull] Device opened\n");
    netif_start_queue(dev);
    request_irq(dev->irq, snull_interrupt, 0, "snull", dev);

    timer_setup(&snull_timer, snull_timer_function, 0);
    mod_timer(&snull_timer, jiffies + msecs_to_jiffies(30000));

    return 0;
}

static int snull_stop(struct net_device *dev) {
    printk(KERN_INFO "[snull] Device stopped\n");
    netif_stop_queue(dev);
    free_irq(dev->irq, dev);
    del_timer_sync(&snull_timer);
    return 0;
}

static netdev_tx_t snull_xmit(struct sk_buff *skb, struct net_device *dev) {
    printk(KERN_INFO "[snull] Transmitting packet\n");
    dev_kfree_skb(skb);
    return NETDEV_TX_OK;
}

static struct net_device_ops snull_ops = {
    .ndo_open = snull_open,
    .ndo_stop = snull_stop,
    .ndo_start_xmit = snull_xmit,
};

static void snull_setup(struct net_device *dev) {
    dev->netdev_ops = &snull_ops;
    ether_setup(dev);
    dev->flags |= IFF_NOARP;
    dev->features |= NETIF_F_HW_CSUM;
    dev->irq = 10;
}

static int __init snull_init(void) {
    int ret;
    snull_dev = alloc_netdev(0, "snull0", NET_NAME_UNKNOWN, snull_setup);
    if (!snull_dev) {
        printk(KERN_ERR "[snull] Failed to allocate network device\n");
        return -ENOMEM;
    }

    ret = register_netdev(snull_dev);
    if (ret) {
        printk(KERN_ERR "[snull] Failed to register network device\n");
        free_netdev(snull_dev);
        return ret;
    }

    printk(KERN_INFO "[snull] Network device registered as snull0\n");
    return 0;
}

static void __exit snull_exit(void) {
    unregister_netdev(snull_dev);
    free_netdev(snull_dev);
    del_timer_sync(&snull_timer);
    printk(KERN_INFO "[snull] Network device unregistered\n");
}

module_init(snull_init);
module_exit(snull_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("Your Name");
MODULE_DESCRIPTION("Simulated Null Network Device with Timer-based Interrupts");

1. 상부 수신 프로그램 (User-space)

이제 snull0 인터페이스를 통해 데이터를 수신하는 **유저 공간 프로그램**을 작성합니다.

📌 snull_receiver.c

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <netpacket/packet.h>
#include <net/ethernet.h>
#include <sys/socket.h>
#include <sys/ioctl.h>
#include <net/if.h>

#define BUFFER_SIZE 2048

int main() {
    int sockfd;
    char buffer[BUFFER_SIZE];
    struct sockaddr_ll addr;
    struct ifreq ifr;

    // RAW 소켓 생성
    sockfd = socket(AF_PACKET, SOCK_RAW, htons(ETH_P_ALL));
    if (sockfd < 0) {
        perror("Socket creation failed");
        return EXIT_FAILURE;
    }

    // 인터페이스 설정
    memset(&ifr, 0, sizeof(ifr));
    strncpy(ifr.ifr_name, "snull0", IFNAMSIZ);
    if (ioctl(sockfd, SIOCGIFINDEX, &ifr) < 0) {
        perror("IOCTL failed");
        close(sockfd);
        return EXIT_FAILURE;
    }

    // 소켓 바인딩
    memset(&addr, 0, sizeof(addr));
    addr.sll_family = AF_PACKET;
    addr.sll_protocol = htons(ETH_P_ALL);
    addr.sll_ifindex = ifr.ifr_ifindex;

    if (bind(sockfd, (struct sockaddr*)&addr, sizeof(addr)) < 0) {
        perror("Bind failed");
        close(sockfd);
        return EXIT_FAILURE;
    }

    printf("Listening for packets on snull0...\n");

    while (1) {
        int num_bytes = recvfrom(sockfd, buffer, BUFFER_SIZE, 0, NULL, NULL);
        if (num_bytes > 0) {
            printf("Received packet of %d bytes\n", num_bytes);
        }
    }

    close(sockfd);
    return 0;
}
