네트워크 드라이브 I

Snull(Simulated Null Network Device)처럼 **ifconfig로 등록되는 가상 네트워크 드라이브**를 커널 모듈로 작성하려면, **TUN/TAP 인터페이스 또는 가상 이더넷 장치(virtual ethernet device)를 활용한 커널 모듈**을 만들어야 합니다. 수행과정은 다음과 같습니다.

  • snull 같은 네트워크 인터페이스를 만들기 위해 가상 네트워크 장치 등록
  • ifconfig snull0 up으로 활성화되도록 구현
  • 네트워크 패킷을 처리할 Net Device Driver 개발

1. 가상 네트워크 장치 커널 모듈 작성

네트워크 장치를 생성하려면 **struct net_device를 사용하여 등록**해야 합니다.

네트워크 장치 드라이버 커널 모듈 (snull.c) 

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/netdevice.h>
#include <linux/etherdevice.h>
#include <linux/skbuff.h>
#include <linux/ip.h>
#include <linux/tcp.h>
#include <linux/udp.h>

static struct net_device *snull_dev;

static int snull_open(struct net_device *dev) {
    printk(KERN_INFO "[snull] Device opened\n");
    netif_start_queue(dev);
    return 0;
}

static int snull_stop(struct net_device *dev) {
    printk(KERN_INFO "[snull] Device stopped\n");
    netif_stop_queue(dev);
    return 0;
}

static netdev_tx_t snull_xmit(struct sk_buff *skb, struct net_device *dev) {
    printk(KERN_INFO "[snull] Transmitting packet\n");
    dev_kfree_skb(skb);
    return NETDEV_TX_OK;
}

static struct net_device_ops snull_ops = {
    .ndo_open = snull_open,
    .ndo_stop = snull_stop,
    .ndo_start_xmit = snull_xmit,
};

static void snull_setup(struct net_device *dev) {
    dev->netdev_ops = &snull_ops;
    ether_setup(dev);
    dev->flags |= IFF_NOARP;
    dev->features |= NETIF_F_HW_CSUM;
}

static int __init snull_init(void) {
    int ret;
    snull_dev = alloc_netdev(0, "snull0", NET_NAME_UNKNOWN, snull_setup);
    if (!snull_dev) {
        printk(KERN_ERR "[snull] Failed to allocate network device\n");
        return -ENOMEM;
    }

    ret = register_netdev(snull_dev);
    if (ret) {
        printk(KERN_ERR "[snull] Failed to register network device\n");
        free_netdev(snull_dev);
        return ret;
    }

    printk(KERN_INFO "[snull] Network device registered as snull0\n");
    return 0;
}

static void __exit snull_exit(void) {
    unregister_netdev(snull_dev);
    free_netdev(snull_dev);
    printk(KERN_INFO "[snull] Network device unregistered\n");
}

module_init(snull_init);
module_exit(snull_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("Your Name");
MODULE_DESCRIPTION("Simulated Null Network Device");

2. 커널 모듈 컴파일 & 로드

1) Makefile 작성

obj-m += snull.o
KDIR := /lib/modules/$(shell uname -r)/build
PWD := $(shell pwd)

all:
    make -C $(KDIR) M=$(PWD) modules

clean:
    make -C $(KDIR) M=$(PWD) clean

2) 빌드 & 모듈 로드 

make
sudo insmod snull.ko

6. 인터럽터를 통한 수신코드

외부로 부터 패킷이 수신되면, 수신된 패킷은 인터럽트를 통해 내부로 전달 된다.

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/netdevice.h>
#include <linux/etherdevice.h>
#include <linux/skbuff.h>
#include <linux/ip.h>
#include <linux/tcp.h>
#include <linux/udp.h>
#include <linux/interrupt.h>

static struct net_device *snull_dev;

static irqreturn_t snull_interrupt(int irq, void *dev_id) {
    struct net_device *dev = (struct net_device *)dev_id;
    struct sk_buff *skb;
    struct iphdr *iph;

    skb = netdev_alloc_skb(dev, 1500);

    /* 하드웨어 수신 코드 추가 */

    /***********************/


    if (!skb)
        return IRQ_HANDLED;

    skb_put(skb, 1500);
    skb->protocol = eth_type_trans(skb, dev);
    iph = ip_hdr(skb);

    printk(KERN_INFO "[snull] Packet received via interrupt\n");
    netif_rx(skb);

    return IRQ_HANDLED;
}


static int snull_open(struct net_device *dev) {
    int ret;

    printk(KERN_INFO "[snull] Device opened\n");
    netif_start_queue(dev);

    ret = request_irq(dev->irq, snull_interrupt, 0, "snull", dev);
    if (ret) {
        pr_err("snull: Failed to request IRQ %d, error %d\n", dev->irq, ret);
        return ret;
    }
    return 0;
}


static int snull_stop(struct net_device *dev) {
    printk(KERN_INFO "[snull] Device stopped\n");
    netif_stop_queue(dev);
    free_irq(dev->irq, dev);
    return 0;
}

static netdev_tx_t snull_xmit(struct sk_buff *skb, struct net_device *dev) {
    printk(KERN_INFO "[snull] Transmitting packet\n");
    dev_kfree_skb(skb);
    return NETDEV_TX_OK;
}

static struct net_device_ops snull_ops = {
    .ndo_open = snull_open,
    .ndo_stop = snull_stop,
    .ndo_start_xmit = snull_xmit,
};

static void snull_setup(struct net_device *dev) {
    dev->netdev_ops = &snull_ops;
    ether_setup(dev);
    dev->flags |= IFF_NOARP;
    dev->features |= NETIF_F_HW_CSUM;
    dev->irq = 10;
}

static int __init snull_init(void) {
    int ret;
    snull_dev = alloc_netdev(0, "snull0", NET_NAME_UNKNOWN, snull_setup);
    if (!snull_dev) {
        printk(KERN_ERR "[snull] Failed to allocate network device\n");
        return -ENOMEM;
    }

    ret = register_netdev(snull_dev);
    if (ret) {
        printk(KERN_ERR "[snull] Failed to register network device\n");
        free_netdev(snull_dev);
        return ret;
    }

    printk(KERN_INFO "[snull] Network device registered as snull0\n");
    return 0;
}

static void __exit snull_exit(void) {
    unregister_netdev(snull_dev);
    free_netdev(snull_dev);
    printk(KERN_INFO "[snull] Network device unregistered\n");
}

module_init(snull_init);
module_exit(snull_exit);
